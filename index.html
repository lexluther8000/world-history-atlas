const { app, BrowserWindow, WebContentsView, ipcMain } = require("electron"); const path = require("path"); let mainWindow; let browserView; let browserVisible = false; let browserMinimized = false; /* ========================================== BROWSER SIZE ========================================== */ const DEFAULT_WIDTH = 900; const DEFAULT_HEIGHT = 650; /* ========================================== CREATE MAIN WINDOW ========================================== */ function createMainWindow() { mainWindow = new BrowserWindow({ width: 1200, height: 800, minWidth: 800, minHeight: 600, backgroundColor: "#ffffff", webPreferences: { preload: path.join( __dirname, "preload.js" ), contextIsolation: true, nodeIntegration: false } }); mainWindow.loadFile( path.join( __dirname, "index.html" ) ); /* * Create the actual browser. */ browserView = new WebContentsView({ webPreferences: { contextIsolation: true, nodeIntegration: false, sandbox: true, javascript: true, webSecurity: true } }); /* * Add the browser to the main * application window. */ mainWindow.contentView.addChildView( browserView ); /* * Don't show it until the user * presses Open Search. */ browserView.setVisible(false); /* * Navigation events. */ browserView.webContents.on( "did-navigate", (event, url) => { send( "browser-url", url ); updateNavigation(); } ); browserView.webContents.on( "did-navigate-in-page", (event, url) => { send( "browser-url", url ); updateNavigation(); } ); /* * Page title. */ browserView.webContents.on( "page-title-updated", (event, pageTitle) => { send( "browser-title", pageTitle ); } ); /* * Loading failed. */ browserView.webContents.on( "did-fail-load", ( event, errorCode, errorDescription, validatedURL ) => { /* * -3 = navigation was cancelled. * This isn't a real error. */ if (errorCode === -3) { return; } send( "browser-error", { errorCode, errorDescription, url: validatedURL } ); } ); /* * If a website asks to open a new * window, keep it in our ONE tab. */ browserView.webContents.setWindowOpenHandler( ({ url }) => { if (url) { browserView.webContents.loadURL( url ); } return { action: "deny" }; } ); /* * Also catch target="_blank" style * navigation where possible. */ browserView.webContents.on( "will-navigate", (event, url) => { /* * Electron normally handles this * automatically. We intentionally * leave the navigation alone. */ } ); /* * Clean up. */ mainWindow.on( "closed", () => { browserView = null; mainWindow = null; } ); } /* ========================================== SEND MESSAGE TO INDEX ========================================== */ function send( channel, data ) { if (!mainWindow) { return; } mainWindow.webContents.send( channel, data ); } /* ========================================== UPDATE NAVIGATION BUTTONS ========================================== */ function updateNavigation() { if ( !browserView || !mainWindow ) { return; } send( "navigation-state", { back: browserView.webContents.canGoBack(), forward: browserView.webContents.canGoForward() } ); } /* ========================================== POSITION BROWSER ========================================== */ function positionBrowser(bounds) { if ( !browserView || !browserVisible || browserMinimized ) { return; } /* * bounds describes the complete * browser popup in the renderer. * * The first 122px are the browser * toolbar. */ const toolbarHeight = 122; const x = Math.round( Number(bounds.x) || 0 ); const y = Math.round( Number(bounds.y) || 0 ); const width = Math.max( 350, Math.round( Number(bounds.width) || DEFAULT_WIDTH ) ); const height = Math.max( 250, Math.round( Number(bounds.height) || DEFAULT_HEIGHT ) ); browserView.setBounds({ x: x, y: y + toolbarHeight, width: width, height: Math.max( 100, height - toolbarHeight ) }); } /* ========================================== OPEN BROWSER ========================================== */ ipcMain.handle( "open-browser", () => { if ( !browserView || !mainWindow ) { return; } browserVisible = true; browserMinimized = false; browserView.setVisible( true ); /* * Tell index.html that it can * position the browser. */ send( "browser-open", true ); /* * Load Bing the first time. */ if ( !browserView.webContents .getURL() ) { browserView.webContents.loadURL( "https://www.bing.com/" ); } } ); /* ========================================== CLOSE BROWSER ========================================== */ ipcMain.handle( "close-browser", () => { browserVisible = false; browserMinimized = false; if (browserView) { browserView.setVisible( false ); } send( "browser-close", true ); } ); /* ========================================== MINIMIZE ========================================== */ ipcMain.handle( "minimize-browser", () => { if (!browserView) { return; } browserMinimized = !browserMinimized; browserView.setVisible( !browserMinimized ); send( "browser-minimized", browserMinimized ); } ); /* ========================================== SET BROWSER POSITION ========================================== */ ipcMain.handle( "set-browser-bounds", (event, bounds) => { positionBrowser( bounds ); } ); /* ========================================== SEARCH BING ========================================== */ ipcMain.handle( "search", (event, query) => { if ( !browserView || !query ) { return; } const searchURL = "https://www.bing.com/search?q=" + encodeURIComponent( query ); browserView.webContents.loadURL( searchURL ); } ); /* ========================================== NAVIGATE TO WEBSITE ========================================== */ ipcMain.handle( "navigate", (event, input) => { if ( !browserView || !input ) { return; } let value = String(input).trim(); if (!value) { return; } /* * If the user typed something like: * * snokido.com * * turn it into: * * https://snokido.com */ if ( !value.startsWith( "http://" ) && !value.startsWith( "https://" ) ) { value = "https://" + value; } browserView.webContents.loadURL( value ); } ); /* ========================================== BACK ========================================== */ ipcMain.handle( "back", () => { if ( browserView && browserView.webContents.canGoBack() ) { browserView.webContents.goBack(); } } ); /* ========================================== FORWARD ========================================== */ ipcMain.handle( "forward", () => { if ( browserView && browserView.webContents.canGoForward() ) { browserView.webContents.goForward(); } } ); /* ========================================== REFRESH ========================================== */ ipcMain.handle( "refresh", () => { if (browserView) { browserView.webContents.reload(); } } ); /* ========================================== HOME ========================================== */ ipcMain.handle( "home", () => { if (browserView) { browserView.webContents.loadURL( "https://www.bing.com/" ); } } ); /* ========================================== APP READY ========================================== */ app.whenReady().then(() => { createMainWindow(); app.on( "activate", () => { if ( BrowserWindow .getAllWindows() .length === 0 ) { createMainWindow(); } } ); }); /* ========================================== CLOSE APP ========================================== */ app.on( "window-all-closed", () => { if ( process.platform !== "darwin" ) { app.quit(); } } );
